<script>
function updateRefrescoPrice(id) {
    var sizeSelect = document.getElementById('size-refresco-' + id);
    var selectedOption = sizeSelect.options[sizeSelect.selectedIndex];
    var price = selectedOption.getAttribute('data-price');
    document.getElementById('price-refresco-' + id).innerText = 'S/ ' + price;
}

function addRefresco(name, id, btn) {
    var sizeSelect = document.getElementById('size-refresco-' + id);
    var tempSelect = document.getElementById('temp-refresco-' + id);

    var size = sizeSelect.value;
    var price = parseFloat(sizeSelect.options[sizeSelect.selectedIndex].getAttribute('data-price'));
    var temp = tempSelect.value;

    var finalName = name + ' (' + size + ' - ' + temp + ')';
    cartApp.addItem(finalName, price, 'https://images.unsplash.com/photo-1513364776144-60967b0f800f?auto=format&fit=crop&w=400&q=80', 'Refresco', btn);
}
</script>
